``` r
# Load historical price data

# Option 1: interactively choose a file
#file_path <- file.choose()

# Option 2: set path directly
#file_path <- "facebook.csv"  # replace with your CSV file path
stk <- read_csv("facebook.csv", show_col_types = FALSE)

# Ensure the date column is parsed correctly
# Assuming the first column is Date
stk <- stk %>%
  rename(Date = 1) %>%
  mutate(Date = as.Date(Date)) %>%
  arrange(Date)
```

``` r
# Calculate moving averages
stk <- stk %>%
  mutate(
    MA10 = SMA(Close, n = 10),
    MA50 = SMA(Close, n = 50)
  ) %>%
  drop_na()
```

``` r
# Generate trading signals and compute profit
stk <- stk %>%
  mutate(
    Shares = ifelse(MA10 > MA50, 1, 0),
    Close1 = lead(Close),       # next day's close
    Profit = ifelse(!is.na(Close1) & Shares == 1,
                    Close1 - Close, 0)
  )
```

``` r
# Plot daily profit with zero baseline
ggplot(stk, aes(x = Date, y = Profit)) +
  geom_line() +
  geom_hline(yintercept = 0, linetype = "dashed", color = "red") +
  labs(title = "Daily Profit", x = "Date", y = "Profit")
```

``` r
# Calculate cumulative wealth
stk <- stk %>%
  mutate(
    wealth = cumsum(Profit)
  )
```

``` r
# Plot cumulative wealth over time
total_money <- last(stk$wealth)

ggplot(stk, aes(x = Date, y = wealth)) +
  geom_line() +
  labs(
    title = paste("Total money you win is", round(total_money, 2)),
    x = "Date",
    y = "Cumulative Wealth"
  )
```
